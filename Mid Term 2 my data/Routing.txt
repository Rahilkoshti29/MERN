Step 1 : main.jsx


import { createRoot } from 'react-dom/client'
import "bootstrap/dist/css/bootstrap.css"
import { createBrowserRouter, RouterProvider } from 'react-router'
import App from './App.jsx'
import HomePage from './pages/HomePage.jsx'
import FoodsPage from './pages/FoodsPage.jsx'
import PageNotFound from './pages/PageNotFound.jsx'
const myRoutes = createBrowserRouter([
 {
 path: "/",
 element: <App />,
 children: [
 { path: "", index: true, element: <HomePage /> },
 { path: "foods", element: <FoodsPage /> },
 ],
 },
 { path: "*", element: <PageNotFound /> },
])
createRoot(document.getElementById('root')).render(
 <RouterProvider router={myRoutes} />
)

Step 7 : pages/FoodsPage.jsx


import { useState } from 'react'
import FoodCard from '../components/FoodCard'
const foods = [
 { id: 1, name: "Pizza", category: "Italian", price: 250 },
 { id: 2, name: "Dosa", category: "South Indian", price: 100 },
 { id: 3, name: "Pasta", category: "Italian", price: 200 },
 { id: 4, name: "Sandwich", category: "Fast Food", price: 120 },
]
const FoodsPage = () => {
 const [search, setSearch] = useState("")
 const filtered = foods.filter((x) =>
 x.name.toLowerCase().includes(search.toLowerCase())
 )
 return (
 <div>
 <h2>Foods</h2>
 <input
 type="text"
 className="form-control mb-3"
placeholder="Search by name..."
 value={search}
 onChange={(e) => setSearch(e.target.value)}
 />
 {filtered.length === 0 ? (
 <p>No food items found</p>
 ) : (
 filtered.map((x) => (
 <FoodCard key={x.id} name={x.name} category={x.category} price={x.price} />
 ))
 )}
 </div>
 )
}
export default FoodsPage
